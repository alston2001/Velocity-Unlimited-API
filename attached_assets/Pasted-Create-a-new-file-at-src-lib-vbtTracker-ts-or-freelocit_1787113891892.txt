export interface VBTState {
  velocity: number;   // Current vertical velocity in m/s  position: number;   // Cumulative vertical displacement in meters  isCalibrated: boolean;
}
export class VBTTracker {
  private plateDiameterM: number;
  private metersPerPixel: number;
  private calibrated: boolean = false;
  
  // State vector x = [position (m), velocity (m/s)]^Tprivate x: [number, number] = [0, 0];
  
  // State Covariance Matrix P (2x2)private P: [[number, number], [number, number]] = [
    [0.1, 0.0],
    [0.0, 0.1]
  ];
  
  private cumVisualPos: number = 0;
  
  // Noise Covariance Tuning Parametersprivate sigmaA: number = 0.35;    // Accelerometer noise std dev (m/s^2)private sigmaVis: number = 0.003; // Optical flow position noise std dev (m)constructor(plateDiameterM: number = 0.45, defaultPlatePx: number = 300) {
    this.plateDiameterM = plateDiameterM;
    this.metersPerPixel = plateDiameterM / defaultPlatePx;
  }
/**
 
Dynamic scale setup using plate bounding box pixel height (450mm reference).

*/

public calibrateScale(plateHeightPx: number): boolean {

if (plateHeightPx > 20) {

this.metersPerPixel = this.plateDiameterM / plateHeightPx;

this.calibrated = true;

return true;

}

return false;

}

/**
 
Resets state vectors and covariance matrices between sets/reps.

*/

public reset(): void {

this.x = [0.0, 0.0];

this.P = [[0.1, 0.0], [0.0, 0.1]];

this.cumVisualPos = 0.0;

}

/**
 
Core Sensor Fusion Execution Step (1D Kalman Filter).
@param deltaYPixels Y-axis pixel movement from visual tracking/flow
@param accelY Gravity-compensated Y-axis acceleration from IMU (m/s^2)
@param dt Time delta between frames in seconds (default ~60 FPS)
@returns Current VBTState containing velocity (m/s) and position (m)

*/

public processFrame(deltaYPixels: number, accelY: number, dt: number = 0.0166): VBTState {

if (dt <= 0) dt = 0.0166;

// 1. Visual Measurement Update (Invert sign: visual downward = physical upward)
const deltaYMeters = -deltaYPixels * this.metersPerPixel;
this.cumVisualPos += deltaYMeters;
// 2. Kalman Filter Prediction Step (Driven by Accelerometer control input)
// x_pred = F * x + B * accelY
const xPred0 = this.x[0] + dt * this.x[1] + 0.5 * Math.pow(dt, 2) * accelY;
const xPred1 = this.x[1] + dt * accelY;
// Discrete Process Noise Q = (sigma_a^2) * [0.25*dt^4, 0.5*dt^3; 0.5*dt^3, dt^2]
const qBase = Math.pow(this.sigmaA, 2);
const Q00 = qBase * 0.25 * Math.pow(dt, 4);
const Q01 = qBase * 0.5 * Math.pow(dt, 3);
const Q11 = qBase * Math.pow(dt, 2);
// P_pred = F * P * F^T + Q
const P00 = this.P[0][0] + dt * (this.P[1][0] + this.P[0][1]) + Math.pow(dt, 2) * this.P[1][1] + Q00;
const P01 = this.P[0][1] + dt * this.P[1][1] + Q01;
const P10 = this.P[1][0] + dt * this.P[1][1] + Q01;
const P11 = this.P[1][1] + Q11;
// 3. Kalman Filter Measurement Correction Step (Driven by Visual Position)
const z = this.cumVisualPos;
const yTilde = z - xPred0; // Measurement Residual
const S = P00 + Math.pow(this.sigmaVis, 2); // Innovation Covariance (H * P * H^T + R)
// Optimal Kalman Gain K = P_pred * H^T / S  (where H = [1, 0])
const K0 = P00 / S;
const K1 = P10 / S;
// Corrected State Update: x = x_pred + K * yTilde
this.x[0] = xPred0 + K0 * yTilde;
this.x[1] = xPred1 + K1 * yTilde;
// Corrected Covariance Update: P = (I - K * H) * P_pred
this.P[0][0] = (1 - K0) * P00;
this.P[0][1] = (1 - K0) * P01;
this.P[1][0] = P10 - K1 * P00;
this.P[1][1] = P11 - K1 * P01;
return {
  velocity: this.x[1],
  position: this.x[0],
  isCalibrated: this.calibrated,
};
}
